function NMN_Slash(a, name)
	if(string.lower(name) == "all") then
		if(GetFriendCount("HateFriend") > 0) then
			local count = 0;
			while(GetFriendCount("HateFriend")>0) do
				DelFriend("HateFriend", GetFriendInfo("HateFriend", 1));
				count = count + 1;
			end
			DEFAULT_CHAT_FRAME:AddMessage(count .. "件削除しました");
		else
			DEFAULT_CHAT_FRAME:AddMessage("1人も登録されていません");
		end
	elseif(tonumber(name) ~= nil) then
		DelFriend("HateFriend", GetFriendInfo("HateFriend", tonumber(name)));
	else
		DelFriend("HateFriend", name);
	end
end;

Kitty.Person.constructor = function(self, name, age)
	-- データメンバ
	local obj = {};
	obj.name = name;
	obj.age = age;

	-- デストラクタの指定
	assert(newproxy, "newproxy does not support.");
	obj.__userdata = newproxy(true);
	getmetatable(obj.__userdata).__gc = function() obj:destructor(); end;
	-- 関数の付与、文字列表現サポート
	return setmetatable(obj, {__index=Kitty.Person, __tostring=Kitty.Person.toString});

		ユーザーが開こうとしているドロップダウンの第１レベルの項目(親項目)。

--[[
-- データ項目の更新(再帰)
Kitty.updateSettingData = function(oldStoredValue, newInitialValue)
	if(type(oldStoredValue) == "table" and type(newInitialValue) == "table") then
		-- どちらもテーブル
		local result = {};
		for key, value in pairs(newInitialValue) do
			local newValue = newInitialValue[key]
			local oldValue = oldStoredValue[key]
			result[key] = Kitty.updateSettingData(oldValue, newValue);	-- 再帰呼出し
		end
		return result;
	else
		if(oldStoredValue == nil) then
			return newInitialValue;
		else
			return oldStoredValue;
		end
	end
end;
--]]
